# ============================================================
# MAIN
# ============================================================
_start:
   #  --- Khởi tạo LCD ---
    li   x10, 0x02            # Trở về vị trí đầu dòng(Return home)
    jal  x1, lcd_write_com
    li   x10, 0x38            # Đặt chức năng giao tiếp 8 bit,2 dòng
    jal  x1, lcd_write_com
    li   x10, 0x01            # Xóa màn hình(Clear display)
    jal  x1, lcd_write_com
    li   x10, 0x0C            # Hiện màn hình,tắt con trỏ
    jal  x1, lcd_write_com
    li   x10, 0x06            # Dịch con trỏ sang phải(khi ghi/đọc data)
    jal  x1, lcd_write_com
    li   x10, 0x80            # Chuyển con trỏ về đầu dòng 1
    jal  x1, lcd_write_com

loop_main:
    # --- Đọc switch SW[3:0] ---
    jal  x2, read_switch      # trả về x10 = 0–15
    add  x14, x0, x10         # lưu lại
    add  x8, x0, x10
    lui  x12, 0x10000
    sw   x14, 0(x12)
   
    # --- Hàng 1: HEX + DEC ---
    li   x10, 0x80
    jal  x1, lcd_write_com
    jal  x4, hex_to_dec
    
    # --- Hàng 2: BIN ---
    li   x10, 0xC0
    jal  x1, lcd_write_com
    jal  x5, hex_to_binary
    j loop_main

# ============================================================
# SUB: hex_to_dec
# In: x14 = giá trị (0–15)
# ============================================================
hex_to_dec:
    # --- In "HEX:" ---
    li x10, 'H'
    jal x1, lcd_write_char
    li x10, 'E'
    jal x1, lcd_write_char
    li x10, 'X'
    jal x1, lcd_write_char
    li x10, ':'
    jal x1, lcd_write_char
    li x10, ' '
    jal x1, lcd_write_char

    # --- In ký tự HEX ---
    add  x10, x0, x14
    li   x11, 10
    blt  x10, x11, h_digit
    addi x10, x10, 55
    j    h_done
h_digit:
    addi x10, x10, 48
h_done:
    jal  x1, lcd_write_char

    # --- In " DEC:" ---
    li x10, ' '
    jal x1, lcd_write_char
    li x10, 'D'
    jal x1, lcd_write_char
    li x10, 'E'
    jal x1, lcd_write_char
    li x10, 'C'
    jal x1, lcd_write_char
    li x10, ':'
    jal x1, lcd_write_char
    li x10, ' '
    jal x1, lcd_write_char

    # --- In giá trị DEC ---
    add  x10, x0, x14         # dividend
    addi x11, x0, 10              # divisor

    bltu  x10, x11, only_one
    addi x10, x0, '1'
    jal  x1, lcd_write_char
    addi x11, ,x0, 10
    sub  x15, x14, x11
    addi x10, x15, 0x30
    jal  x1, lcd_write_char
    j    dec_done
only_one:
    addi x10, x0, '0'
    jal  x1, lcd_write_char
    addi x10, x14, 0x30
    jal  x1, lcd_write_char
    
dec_done:
    jalr x0, x4, 0

# ============================================================
# SUB: hex_to_binary
# In: x10 = giá trị (0–15)
# ============================================================
hex_to_binary:
    li   x10, 'B'
    jal x1, lcd_write_char
    li   x10, 'I'
    jal x1, lcd_write_char
    li   x10, 'N'
    jal x1, lcd_write_char
    li   x10, ':'
    jal x1, lcd_write_char
    li   x10, ' '
    jal x1, lcd_write_char
    #li   x11, 3
bin3:    
    andi x21, x8, 8
    addi x20, x0, 8
    beq  x20, x21, print_1_to_bin3
    li   x10, '0'
    jal  x1, lcd_write_char
bin2:
    andi x21, x8, 4
    addi x20, x0, 4
    beq  x20, x21, print_1_to_bin2
    li   x10, '0'
    jal  x1, lcd_write_char
bin1:
    andi x21, x8, 2
    addi x20, x0, 2
    beq  x20, x21, print_1_to_bin1
    li   x10, '0'
    jal  x1, lcd_write_char
bin0:
    andi x21, x8, 1
    addi x20, x0, 1
    beq  x20, x21, print_1_to_bin0
    li   x10, '0'
    jal  x1, lcd_write_char
    jalr x0, x5, 0
    
print_1_to_bin3:
    li   x10, '1'
    jal  x1, lcd_write_char
    j bin2 
print_1_to_bin2:
    li   x10, '1'
    jal  x1, lcd_write_char
    j bin1
print_1_to_bin1:
    li   x10, '1'
    jal  x1, lcd_write_char
    j bin0
print_1_to_bin0:
    li   x10, '1'
    jal  x1, lcd_write_char
    jalr x0, x5, 0
    

# ============================================================
# uint32_t read_switch(void)
# return: x10 = giá trị SW[16:0]
# ============================================================
read_switch:
    lui  x11, 0x10010         # x11 = 0x1001_0000 (địa chỉ switch base)
    lw   x10, 0(x11)          # đọc dữ liệu 32-bit từ switch

    # Giữ lại switch[3:0]
    addi x12, x0, 0x0F         
    and  x10, x10, x12        

    jalr x0, x2, 0            # return
    
lcd_write_char:
    # x10 = ký tự cần ghi
    lui  x11, 0x10004         # x11 = 0x10004000
    li   x12, 0
    lui  x12, 0x80000 # ON = 1 (bit31)
    ori  x12, x12, 0x00000200 # RS = 1 (bit9)
    or   x12, x12, x10        # thêm ký tự vào 8 bit thấp
    sw   x12, 0(x11)          # ghi LCD với EN=0

    # --- tạo xung EN ---
    ori  x13, x12, 0x00000400 # EN = 1 (bit10)
    sw   x13, 0(x11)
    #jal  x3, delay_2ms
    andi x13, x13, 0xFFFFFBFF # EN = 0
    sw   x13, 0(x11)
    #jal  x3, delay_2ms
    jalr x0, x1, 0            # return
# ============================================================
# void lcd_write_com(uint8_t cmd)
# a0 = command (x10)
# ============================================================
lcd_write_com:
    #addi x10, x0, 0xff
    lui  x11, 0x10004         # x11 = 0x10004000 (LCD base)
    li   x12, 0
    lui  x12, 0x80000         # bit31: ON = 1
    or   x12, x12, x10        # gộp dữ liệu lệnh (8-bit)
    sw   x12, 0(x11)          # ghi LCD, EN=0

    # --- tạo xung EN đúng cách ---
    ori  x13, x12, 0x00000400 # bật EN (bit10 = 1)
    sw   x13, 0(x11)          # ghi LCD với EN=1
    #jal  x3, delay_2ms
    andi x13, x13, 0xFFFFFBFF # tắt EN (bit10 = 0), giữ nguyên các bit khác
    or   x13, x13, x12        # đảm bảo giữ nguyên dữ liệu/ON
    sw   x13, 0(x11)          # ghi LCD với EN=0
    #jal  x3, delay_2ms
    jalr x0, x1, 0            # return
    
delay_2ms:
    lui  x25, 0xD
    addi x25, x25, 0x2F0
delay_2ms_loop:
    addi x25, x25, -1
    addi x0, x0, 0
    beq x25, x0, delay_2ms_done
    j delay_2ms_loop
delay_2ms_done: jalr x0, x3, 0
